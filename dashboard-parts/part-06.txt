  function electionCargoLabel(cargo = electionLiveState.cargo) {
    return electionCargoOptions.find(([id]) => id === cargo)?.[1] || cargo;
  }

  function electionUfLabel(uf = electionLiveState.uf) {
    return electionUfNames[String(uf || 'br').toLowerCase()] || String(uf || '').toUpperCase();
  }

  function electionFormatInteger(value) {
    const number = Number(value || 0);
    return Number.isFinite(number) ? number.toLocaleString('pt-BR', { maximumFractionDigits: 0 }) : '0';
  }

  function electionFormatPercent(value, digits = 2) {
    const number = Number(value || 0);
    return Number.isFinite(number) ? `${number.toLocaleString('pt-BR', { minimumFractionDigits: digits, maximumFractionDigits: digits })}%` : '0%';
  }

  function electionCandidateColor(candidate, index = 0) {
    const signature = `${candidate?.party || ''}|${candidate?.number || ''}|${candidate?.name || ''}`;
    let hash = 0;
    for (let i = 0; i < signature.length; i += 1) hash = ((hash << 5) - hash + signature.charCodeAt(i)) | 0;
    return electionPalette[Math.abs(hash || index) % electionPalette.length];
  }

  function electionCandidatePerson(candidate, index = 0) {
    const name = String(candidate?.name || candidate?.ballotName || 'Candidato').trim();
    const wiki = electionCandidateWiki[name.toUpperCase()] || name;
    return {
      ...candidate,
      name,
      party: candidate?.party || 'Partido não informado',
      color: electionCandidateColor(candidate, index),
      photo: `/api/public-figure-photo?name=${encodeURIComponent(name)}&wiki=${encodeURIComponent(wiki)}`
    };
  }

  async function electionFetchJson(url) {
    const response = await fetch(url, { cache: 'no-store' });
    const payload = await response.json();
    if (!response.ok || !payload.ok) throw new Error(payload.detail || payload.error || 'Falha ao carregar dados eleitorais.');
    return payload;
  }

  function electionQuery(params) {
    const search = new URLSearchParams();
    Object.entries(params).forEach(([key, value]) => {
      if (value !== undefined && value !== null && value !== '') search.set(key, value);
    });
    return `/api/election-results?${search.toString()}`;
  }

  function electionCountdown() {
    const target = new Date('2026-10-04T08:00:00-03:00').getTime();
    const diff = target - Date.now();
    if (diff <= 0) return 'Dia da votação';
    const days = Math.floor(diff / 86400000);
    const hours = Math.floor((diff % 86400000) / 3600000);
    return `${days}d ${hours}h`;
  }

  function electionMapCoordinates(geometry) {
    if (!geometry) return [];
    if (geometry.type === 'Polygon') return geometry.coordinates;
    if (geometry.type === 'MultiPolygon') return geometry.coordinates.flat();
    return [];
  }

  function electionMapPath(rings, bounds, width, height) {
    const [minX, minY, maxX, maxY] = bounds;
    const pad = 18;
    const scale = Math.min((width - pad * 2) / Math.max(.0001, maxX - minX), (height - pad * 2) / Math.max(.0001, maxY - minY));
    const offsetX = (width - (maxX - minX) * scale) / 2;
    const offsetY = (height - (maxY - minY) * scale) / 2;
    return rings.map((ring) => ring.map(([lon, lat], index) => {
      const x = offsetX + (lon - minX) * scale;
      const y = height - (offsetY + (lat - minY) * scale);
      return `${index ? 'L' : 'M'}${x.toFixed(2)},${y.toFixed(2)}`;
    }).join(' ') + ' Z').join(' ');
  }

  function electionAnimateUpdatedValues(root = document) {
    qsa('[data-election-value]', root).forEach((node) => {
      node.classList.remove('is-election-value-updated');
      void node.offsetWidth;
      node.classList.add('is-election-value-updated');
    });
  }

  async function electionLoadCore({ preserveGranular = false } = {}) {
    const token = ++electionLiveState.requestToken;
    electionLiveState.loading = true;
    electionLiveState.error = '';
    if (!preserveGranular) {
      electionLiveState.municipality = '';
      electionLiveState.zone = '';
      electionLiveState.section = '';
      electionLiveState.municipalities = [];
      electionLiveState.municipalityResult = null;
      electionLiveState.sections = [];
    }
    renderElections2026Shell();

    const shared = { cargo: electionLiveState.cargo, round: electionLiveState.round, uf: electionLiveState.uf };
    try {
      const requests = [
        electionFetchJson(electionQuery({ action: 'status' })),
        electionFetchJson(electionQuery({ action: 'summary', year: 2022, ...shared })),
        electionFetchJson(electionQuery({ action: 'states', year: electionLiveState.year, cargo: electionLiveState.cargo, round: electionLiveState.round }))
      ];
      if (electionLiveState.year === 2026) requests.push(electionFetchJson(electionQuery({ action: 'summary', year: 2026, ...shared })));
      const values = await Promise.all(requests);
      if (token !== electionLiveState.requestToken) return;
      electionLiveState.status = values[0];
      electionLiveState.historical = values[1];
      electionLiveState.states = values[2].states || [];
      electionLiveState.summary = electionLiveState.year === 2026 ? values[3] : values[1];
      electionLiveState.lastUpdated = new Date();
    } catch (error) {
      if (token !== electionLiveState.requestToken) return;
      electionLiveState.error = error.message;
      if (!electionLiveState.historical) electionLiveState.historical = { ok: true, candidates: [], waiting: true };
    } finally {
      if (token !== electionLiveState.requestToken) return;
      electionLiveState.loading = false;
      renderElections2026Shell();
      electionAnimateUpdatedValues($('elections2026'));
      if (electionLiveState.year === 2022 && electionLiveState.uf !== 'br') electionLoadMunicipalities();
      electionConfigureAutoRefresh();
    }
  }

  async function electionLoadMunicipalities() {
    if (electionLiveState.year !== 2022 || electionLiveState.uf === 'br') return;
    electionLiveState.granularLoading = true;
    electionLiveState.granularError = '';
    renderElections2026Shell();
    try {
      const payload = await electionFetchJson(electionQuery({ action: 'municipalities', year: 2022, cargo: electionLiveState.cargo, round: electionLiveState.round, uf: electionLiveState.uf }));
      electionLiveState.municipalities = payload.municipalities || [];
    } catch (error) {
      electionLiveState.granularError = error.message;
      electionLiveState.municipalities = [];
    } finally {
      electionLiveState.granularLoading = false;
      renderElections2026Shell();
    }
  }

  async function electionLoadMunicipality() {
    if (electionLiveState.year !== 2022 || !electionLiveState.municipality) return;
    electionLiveState.granularLoading = true;
    electionLiveState.granularError = '';
    electionLiveState.zone = '';
    electionLiveState.section = '';
    electionLiveState.sections = [];
    renderElections2026Shell();
    try {
      const payload = await electionFetchJson(electionQuery({ action: 'municipality', year: 2022, cargo: electionLiveState.cargo, round: electionLiveState.round, uf: electionLiveState.uf, municipality: electionLiveState.municipality }));
      electionLiveState.municipalityResult = payload;
    } catch (error) {
      electionLiveState.granularError = error.message;
      electionLiveState.municipalityResult = null;
    } finally {
      electionLiveState.granularLoading = false;
      renderElections2026Shell();
    }
  }

  async function electionLoadSections() {
    if (electionLiveState.year !== 2022 || !electionLiveState.municipality) return;
    electionLiveState.granularLoading = true;
    electionLiveState.granularError = '';
    electionLiveState.section = '';
    renderElections2026Shell();
    try {
      const payload = await electionFetchJson(electionQuery({ action: 'sections', year: 2022, cargo: electionLiveState.cargo, round: electionLiveState.round, uf: electionLiveState.uf, municipality: electionLiveState.municipality, zone: electionLiveState.zone }));
      electionLiveState.sections = payload.sections || [];
    } catch (error) {
      electionLiveState.granularError = error.message;
      electionLiveState.sections = [];
    } finally {
      electionLiveState.granularLoading = false;
      renderElections2026Shell();
    }
  }

  function electionOpenCandidate(index) {
    const result = electionLiveState.year === 2026 ? electionLiveState.summary : electionLiveState.historical;
    const candidate = result?.candidates?.[Number(index)];
    const dialog = $('pollDialog');
    const body = $('dialogBody');
    if (!candidate || !dialog || !body) return;
    const person = electionCandidatePerson(candidate, Number(index));
    body.innerHTML = `<article class="space-y-5"><header class="flex items-center gap-4 border-l-4 pl-4" style="border-color:${person.color}">${candidatePhotoMarkup(person, 'xl')}<div class="min-w-0"><span class="text-xs font-bold uppercase tracking-widest text-slate-500">${electionLiveState.year} · ${electionLiveState.round}º turno · ${escapeHtml(electionUfLabel())}</span><h2 class="mt-1 font-display text-4xl uppercase text-slate-950">${escapeHtml(person.name)}</h2><p class="mt-1 text-sm text-slate-500">${escapeHtml(person.party)}${person.number ? ` · número ${escapeHtml(person.number)}` : ''}</p></div></header><div class="grid gap-3 sm:grid-cols-3">${[['Votos', electionFormatInteger(person.votes)], ['Percentual', electionFormatPercent(person.percentage, 2)], ['Abrangência', electionUfLabel()]].map(([label, value]) => `<div class="border border-slate-200 bg-slate-50 p-3"><span class="text-[10px] font-bold uppercase tracking-widest text-slate-500">${escapeHtml(label)}</span><strong class="mt-1 block font-display text-2xl text-slate-950">${escapeHtml(value)}</strong></div>`).join('')}</div><div class="border-l-4 border-blue-500 bg-blue-50 p-4 text-sm leading-relaxed text-blue-950">O valor pertence exclusivamente ao cargo, turno e território selecionados. Alterar o estado, município ou zona muda o universo de votos.</div>${result.sourceUrl ? `<a class="inline-flex min-h-11 items-center bg-emerald-600 px-5 py-3 text-sm font-black uppercase text-white" href="${escapeHtml(result.sourceUrl)}" target="_blank" rel="noopener noreferrer">Abrir arquivo oficial</a>` : ''}</article>`;
    bindPhotoFallbacks(body);
    dialog.showModal();
  }

  function electionBindControls(host) {
    host.querySelector('[data-election-year]')?.addEventListener('change', (event) => {
      electionLiveState.year = Number(event.target.value);
      electionLoadCore();
    });
    host.querySelector('[data-election-cargo]')?.addEventListener('change', (event) => {
      electionLiveState.cargo = event.target.value;
      if (!['president','governor'].includes(electionLiveState.cargo)) electionLiveState.round = 1;
      if (electionLiveState.cargo !== 'president' && electionLiveState.uf === 'br') electionLiveState.uf = 'sp';
      electionLoadCore();
    });
    host.querySelector('[data-election-round]')?.addEventListener('change', (event) => {
      electionLiveState.round = Number(event.target.value);
      electionLoadCore();
    });
    host.querySelector('[data-election-uf]')?.addEventListener('change', (event) => {
      electionLiveState.uf = event.target.value;
      electionLoadCore();
    });
    host.querySelector('[data-election-municipality]')?.addEventListener('change', (event) => {
      electionLiveState.municipality = event.target.value;
      electionLiveState.municipalityResult = null;
      electionLiveState.zone = '';
      electionLiveState.sections = [];
      if (electionLiveState.municipality) electionLoadMunicipality();
      else renderElections2026Shell();
    });
    host.querySelector('[data-election-zone]')?.addEventListener('change', (event) => {
      electionLiveState.zone = event.target.value;
      electionLiveState.sections = [];
      electionLoadSections();
    });
    host.querySelector('[data-election-section]')?.addEventListener('change', (event) => {
      electionLiveState.section = event.target.value;
      renderElections2026Shell();
    });
    host.querySelector('[data-election-refresh]')?.addEventListener('click', () => electionLoadCore({ preserveGranular: true }));
    qsa('[data-election-candidate-index]', host).forEach((button) => button.addEventListener('click', () => electionOpenCandidate(button.dataset.electionCandidateIndex)));
    qsa('[data-election-section-button]', host).forEach((button) => button.addEventListener('click', () => {
      electionLiveState.section = button.dataset.electionSectionButton;
      renderElections2026Shell();
    }));
  }

  async function electionEnsureMap() {
    if (electionLiveState.map) return electionLiveState.map;
    try {
      const response = await fetch('/api/brazil-map', { cache: 'force-cache' });
      if (!response.ok) throw new Error('Mapa indisponível');
      electionLiveState.map = await response.json();
    } catch (error) {
      electionLiveState.map = { type: 'FeatureCollection', features: [] };
    }
    return electionLiveState.map;
  }

  async function electionRenderMap() {
    const host = $('elections2026')?.querySelector('[data-election-map-host]');
    const ranking = $('elections2026')?.querySelector('[data-election-map-ranking]');
    if (!host || !ranking) return;
    const geojson = await electionEnsureMap();
    if (!host.isConnected) return;
    const features = geojson?.features || [];
    const stateMap = new Map(electionLiveState.states.filter((item) => item.available).map((item) => [item.uf.toUpperCase(), item]));
    const allCoordinates = features.flatMap((feature) => electionMapCoordinates(feature.geometry).flat());
    if (!features.length || !allCoordinates.length) {
      host.innerHTML = '<div class="grid min-h-[420px] place-items-center border border-dashed border-slate-300 bg-slate-50 p-6 text-center text-sm text-slate-500">A malha cartográfica não pôde ser carregada. Use a lista estadual ao lado.</div>';
    } else {
      const xs = allCoordinates.map((point) => point[0]);
      const ys = allCoordinates.map((point) => point[1]);
      const bounds = [Math.min(...xs), Math.min(...ys), Math.max(...xs), Math.max(...ys)];
      const width = 640;
      const height = 520;
      host.innerHTML = `<svg data-election-map class="h-auto w-full" viewBox="0 0 ${width} ${height}" role="img" aria-label="Mapa do Brasil com resultados eleitorais por estado">${features.map((feature) => {
        const uf = String(feature.properties?.sigla || feature.properties?.SIGLA || feature.properties?.uf || '').toUpperCase();
        const result = stateMap.get(uf);
        const winner = result?.winner ? electionCandidatePerson(result.winner) : null;
        const fill = winner?.color || '#cbd5e1';
        const label = result?.winner ? `${uf}: ${result.winner.name}, ${electionFormatPercent(result.winner.percentage || 0, 2)}` : `${uf}: resultado indisponível`;
        return `<path class="election-map-state ${electionLiveState.uf.toUpperCase() === uf ? 'is-active' : ''}" data-election-map-uf="${uf.toLowerCase()}" tabindex="0" role="button" aria-label="${escapeHtml(label)}" d="${electionMapPath(electionMapCoordinates(feature.geometry), bounds, width, height)}" fill="${fill}"><title>${escapeHtml(label)}</title></path>`;
      }).join('')}</svg>`;
      qsa('[data-election-map-uf]', host).forEach((path) => {
        const activate = () => {
          electionLiveState.uf = path.dataset.electionMapUf;
          electionLoadCore();
        };
        path.addEventListener('click', activate);
        path.addEventListener('keydown', (event) => {
          if (event.key === 'Enter' || event.key === ' ') {
            event.preventDefault();
            activate();
          }
        });
      });
    }

    const rankedStates = electionLiveState.states.filter((item) => item.available && item.winner)
      .sort((a, b) => Number(b.winner.votes || 0) - Number(a.winner.votes || 0));
    ranking.innerHTML = `<div><span class="text-[10px] font-bold uppercase tracking-widest text-slate-500">Estados com dados</span><strong class="mt-1 block font-display text-xl uppercase text-slate-950">${rankedStates.length} de 27 UFs</strong></div><div class="mt-3 max-h-[380px] space-y-1.5 overflow-y-auto">${rankedStates.map((item) => {
      const person = electionCandidatePerson(item.winner);
      return `<button type="button" data-election-state-button="${item.uf.toLowerCase()}" class="election-drill-button grid w-full grid-cols-[38px_minmax(0,1fr)_auto] items-center gap-2 border border-slate-200 bg-white p-2 text-left"><b class="font-display text-lg text-slate-950">${item.uf}</b><span class="min-w-0"><strong class="block truncate text-xs text-slate-800">${escapeHtml(person.name)}</strong><small class="block truncate text-[10px] uppercase text-slate-500">${escapeHtml(person.party)}</small></span><b class="font-display text-base text-slate-950">${electionFormatPercent(person.percentage || 0, 1)}</b></button>`;
    }).join('') || '<div class="border border-dashed border-slate-300 bg-slate-50 p-4 text-sm text-slate-500">Nenhuma UF com votos disponíveis para este recorte.</div>'}</div>`;
    qsa('[data-election-state-button]', ranking).forEach((button) => button.addEventListener('click', () => {
      electionLiveState.uf = button.dataset.electionStateButton;
      electionLoadCore();
    }));
  }

  function electionConfigureAutoRefresh() {
    if (electionLiveState.autoRefresh) clearInterval(electionLiveState.autoRefresh);
    electionLiveState.autoRefresh = null;
    if (electionLiveState.year !== 2026 || !electionLiveState.status?.live) return;
    electionLiveState.autoRefresh = setInterval(() => {
      if (document.visibilityState === 'visible' && state.view === 'elections2026') electionLoadCore({ preserveGranular: true });
    }, 20000);
  }

  function renderElections2026() {
    electionInstallStyles();
    renderElections2026Shell();
    if (!electionLiveState.status && !electionLiveState.loading) electionLoadCore();
  }

  const electionPreviousShowView = showView;
  showView = function showViewWithElections2026(id) {
    if (id !== 'elections2026') return electionPreviousShowView(id);
    state.view = id;
    qsa('.view-section').forEach((section) => section.classList.toggle('hidden', section.id !== id));
    qsa('[data-view]').forEach((button) => button.classList.toggle('is-active', button.dataset.view === id));
    $('sidebar').classList.remove('translate-x-0');
    $('sidebar').classList.add('-translate-x-full', 'lg:translate-x-0');
    renderElections2026();
    window.scrollTo({ top: 0, behavior: 'smooth' });
    history.replaceState(null, '', '#elections2026');
  };

  window.addEventListener('resize', () => {
    if (state.view === 'elections2026') electionRenderMap();
  });
