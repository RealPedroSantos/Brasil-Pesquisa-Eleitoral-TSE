      svg.appendChild(dateText);
      const instituteText = svgEl('text', { x: xx, y: height - 37, class: 'chart-institute-label', 'text-anchor': 'middle' }, poll.institute.length > 14 ? `${poll.institute.slice(0, 13)}…` : poll.institute);
      svg.appendChild(instituteText);
      const scenarioText = svgEl('text', { x: xx, y: height - 20, class: 'chart-scenario-label', 'text-anchor': 'middle' }, polls.filter((item) => item.date === poll.date).length > 1 ? poll.scenario.replace('Cenário ', 'C') : '');
      svg.appendChild(scenarioText);
    });

    const scenarioMode = (poll) => {
      const text = String(poll?.scenario || '').normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase();
      if (text.includes('espont')) return 'espontanea';
      if (text.includes('estim')) return 'estimulada';
      return 'estimulado';
    };
    const comparableSeriesKey = (poll) => `${poll?.round || 1}|${scenarioMode(poll)}|${Object.keys(poll?.values || {}).sort().join('|')}`;

    candidateIds.forEach((candidateId) => {
      const candidate = candidateById(candidateId);
      const points = [];
      const segments = [];
      let segment = [];

      polls.forEach((poll, index) => {
        if (!Object.hasOwn(poll.values, candidateId)) {
          if (segment.length > 1) segments.push(segment);
          segment = [];
          return;
        }

        const point = { poll, index, value: poll.values[candidateId], x: x(index), y: y(poll.values[candidateId]) };
        const previous = segment.at(-1);
        const repeatedDate = previous && String(previous.poll.date || '') === String(poll.date || '');
        const incompatibleScenario = previous && comparableSeriesKey(previous.poll) !== comparableSeriesKey(poll);

        if (repeatedDate || incompatibleScenario) {
          if (segment.length > 1) segments.push(segment);
          segment = [];
        }

        segment.push(point);
        points.push(point);
      });

      if (segment.length > 1) segments.push(segment);
      if (!candidate || !points.length) return;

      segments.forEach((seriesPoints) => {
        const path = svgEl('path', {
          d: seriesPoints.map((point, index) => `${index ? 'L' : 'M'} ${point.x} ${point.y}`).join(' '),
          fill: 'none', stroke: candidate.color, 'stroke-width': state.round === 2 ? 5 : 3.5,
          'stroke-linecap': 'round', 'stroke-linejoin': 'round', class: 'chart-series-line'
        });
        svg.appendChild(path);
      });

      points.forEach((point) => {
        const hit = svgEl('circle', { cx: point.x, cy: point.y, r: 14, fill: 'transparent', class: 'chart-hit-area', tabindex: '0', role: 'button', 'aria-label': `${candidate.name}, ${formatNumber(point.value)}%, ${point.poll.institute}, ${point.poll.publication}` });
        const dot = svgEl('circle', { cx: point.x, cy: point.y, r: state.round === 2 ? 6.5 : 5.5, fill: candidate.color, stroke: '#020617', 'stroke-width': 3, class: 'chart-point' });
        const show = (event) => showChartTooltip(event, tooltip, candidate, point.poll, point.value);
        hit.addEventListener('pointerenter', show);
        hit.addEventListener('pointermove', show);
        hit.addEventListener('pointerleave', () => tooltip?.classList.add('hidden'));
        hit.addEventListener('focus', (event) => showChartTooltip(event, tooltip, candidate, point.poll, point.value, true));
        hit.addEventListener('blur', () => tooltip?.classList.add('hidden'));
        hit.addEventListener('click', () => openPollDialog(point.poll.id));
        hit.addEventListener('keydown', (event) => { if (event.key === 'Enter' || event.key === ' ') openPollDialog(point.poll.id); });
        svg.appendChild(dot);
        svg.appendChild(hit);
      });

      const latestPoint = points.at(-1);
      const portraitSize = state.round === 2 ? 48 : 34;
      const half = portraitSize / 2;
      const clipId = `${svg.id}-${candidate.id}-portrait`;
      const clip = svgEl('clipPath', { id: clipId });
      clip.appendChild(svgEl('rect', { x: latestPoint.x - half, y: latestPoint.y - half, width: portraitSize, height: portraitSize }));
      defs.appendChild(clip);
      svg.appendChild(svgEl('rect', { x: latestPoint.x - half, y: latestPoint.y - half, width: portraitSize, height: portraitSize, fill: '#0f172a', class: 'chart-candidate-photo-fallback' }));
      svg.appendChild(svgEl('text', { x: latestPoint.x, y: latestPoint.y + 4, fill: '#ffffff', class: 'chart-candidate-initials', 'text-anchor': 'middle' }, candidateInitials(candidate)));
      const portrait = svgEl('image', {
        href: candidatePhoto(candidate), x: latestPoint.x - half, y: latestPoint.y - half,
        width: portraitSize, height: portraitSize, preserveAspectRatio: 'xMidYMid slice',
        'clip-path': `url(#${clipId})`, class: 'chart-candidate-photo'
      });
      portrait.addEventListener('error', () => portrait.remove(), { once: true });
      svg.appendChild(portrait);
      svg.appendChild(svgEl('rect', { x: latestPoint.x - half, y: latestPoint.y - half, width: portraitSize, height: portraitSize, fill: 'none', stroke: candidate.color, 'stroke-width': 3, class: 'chart-candidate-photo-border' }));
    });

    if (!polls.length) {
      svg.appendChild(svgEl('text', { x: width / 2, y: height / 2, class: 'chart-empty-label', 'text-anchor': 'middle' }, 'Nenhuma pesquisa encontrada para este filtro.'));
    }

    if (ranking) {
      const rankingRows = candidateIds.map((id) => {
        const candidate = candidateById(id);
        const poll = polls.slice().reverse().find((item) => Object.hasOwn(item.values, id));
        return candidate && poll ? { candidate, poll, value: poll.values[id] } : null;
      }).filter(Boolean).sort((a, b) => b.value - a.value || a.candidate.name.localeCompare(b.candidate.name));
      ranking.innerHTML = `<div class="chart-ranking-heading"><span>Último dado disponível</span><strong>Ranking com fotos</strong></div>
        <div class="chart-ranking-list">${rankingRows.map((row, index) => `<button type="button" data-poll-id="${row.poll.id}" class="chart-ranking-row" style="--candidate-color:${row.candidate.color}">
          <span class="chart-ranking-position">${index + 1}</span>${candidatePhotoMarkup(row.candidate, 'sm')}
          <span class="min-w-0"><strong>${escapeHtml(row.candidate.name)}</strong><small>${escapeHtml(row.candidate.party)} · ${escapeHtml(row.poll.publication)}</small></span>
          <b>${formatNumber(row.value)}%</b>
        </button>`).join('')}</div>`;
      bindPhotoFallbacks(ranking);
      ranking.querySelectorAll('[data-poll-id]').forEach((button) => button.addEventListener('click', () => openPollDialog(button.dataset.pollId)));
    }

    if (legend) {
      legend.innerHTML = candidateIds.map((id) => {
        const candidate = candidateById(id);
        return `<span class="candidate-legend-item" style="--candidate-color:${candidate.color}">${candidatePhotoMarkup(candidate, 'xs')}<span>${escapeHtml(candidate.name)} · ${escapeHtml(candidate.party)}</span></span>`;
      }).join('');
      bindPhotoFallbacks(legend);
    }
  }

  function showChartTooltip(event, tooltip, candidate, poll, value, keyboard = false) {
    if (!tooltip) return;
    tooltip.innerHTML = `<div class="flex items-center gap-3">${candidatePhotoMarkup(candidate, 'md')}<span><strong class="block font-display text-lg uppercase text-white">${escapeHtml(candidate.name)} · ${formatNumber(value)}%</strong><small class="block text-xs text-slate-400">${escapeHtml(candidate.party)}</small></span></div><span class="mt-3 block text-xs text-slate-300">${escapeHtml(poll.institute)} · ${escapeHtml(poll.publication)}</span><span class="mt-1 block text-xs text-slate-500">${escapeHtml(poll.scenario)} · ${escapeHtml(poll.sample)} · ${escapeHtml(poll.margin)}</span>`;
    bindPhotoFallbacks(tooltip);
    tooltip.classList.remove('hidden');
    const parent = tooltip.parentElement.getBoundingClientRect();
    const x = keyboard ? parent.width / 2 : event.clientX - parent.left;
    const y = keyboard ? 80 : event.clientY - parent.top;
    tooltip.style.left = `${Math.max(8, Math.min(parent.width - 270, x + 12))}px`;
    tooltip.style.top = `${Math.max(8, y - 94)}px`;
  }

    const body = $('pollTableBody');
    if (!body) return;
    const polls = DATA.polls.slice().sort((a, b) => dateValue(b.date) - dateValue(a.date) || a.id.localeCompare(b.id));
    const count = $('pollTableCount');
    count.textContent = `${polls.length} cenários incorporados · cobertura em auditoria`;
    count.title = 'O total representa os resultados já incorporados ao sistema e não a totalidade de registros existentes no TSE.';
    body.innerHTML = polls.map((poll) => `<tr>
      <td><strong>${escapeHtml(poll.publication)}</strong></td>

      <td><div class="flex min-w-[320px] flex-wrap gap-1">${Object.entries(poll.values).map(([id, value]) => { const candidate = candidateById(id); return `<button type="button" data-poll-id="${poll.id}" class="poll-table-candidate">${candidatePhotoMarkup(candidate, 'xs')}<span>${escapeHtml(candidate?.name || id)} <b>${formatNumber(value)}%</b></span></button>`; }).join('')}</div></td>

    bindPhotoFallbacks(body);
    body.querySelectorAll('[data-poll-id]').forEach((button) => button.addEventListener('click', () => openPollDialog(button.dataset.pollId)));
  }
