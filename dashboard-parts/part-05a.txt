  function extendedHemicycleSvg(model) {
    const rows = model.total > 100 ? 13 : 6;
    const weights = Array.from({ length: rows }, (_, index) => index + 5);
    const weightTotal = weights.reduce((sum, value) => sum + value, 0);
    const counts = weights.map((weight) => Math.floor((model.total * weight) / weightTotal));
    let missing = model.total - counts.reduce((sum, value) => sum + value, 0);
    for (let index = counts.length - 1; missing > 0; index = (index - 1 + counts.length) % counts.length) {
      counts[index] += 1;
      missing -= 1;
    }

    const positions = [];
    const minRadius = model.total > 100 ? 118 : 145;
    const maxRadius = 430;
    counts.forEach((count, row) => {
      const radius = rows === 1 ? maxRadius : minRadius + ((maxRadius - minRadius) * row) / (rows - 1);
      for (let index = 0; index < count; index += 1) {
        const angle = count === 1 ? Math.PI / 2 : Math.PI - (Math.PI * index) / (count - 1);
        positions.push({ angle, radius, x: 500 + radius * Math.cos(angle), y: 472 - radius * Math.sin(angle) });
      }
    });
    positions.sort((a, b) => b.angle - a.angle || b.radius - a.radius);

    const seats = model.parties.flatMap((item) => Array.from({ length: item.seats }, () => item));
    const circleRadius = model.total > 100 ? 4.5 : 8.5;
    const circles = positions.map((position, index) => {
      const item = seats[index] || { party: 'Outros' };
      return `<circle cx="${position.x.toFixed(2)}" cy="${position.y.toFixed(2)}" r="${circleRadius}" fill="${extendedColor(item.party)}"><title>${escapeHtml(item.party)}</title></circle>`;
    }).join('');

    return `<svg class="h-auto w-full" viewBox="0 0 1000 500" role="img" aria-label="${escapeHtml(model.title)} com ${model.total} cadeiras distribuídas por partido">${circles}</svg>`;
  }

  function bindExtendedOfficeControls(root = document) {
    bindPhotoFallbacks(root);
    qsa('[data-extended-office-view]', root).forEach((button) => {
      if (button.dataset.extendedBound === '1') return;
      button.dataset.extendedBound = '1';
      button.addEventListener('click', () => showView(button.dataset.extendedOfficeView));
    });
    qsa('[data-overview-jump]', root).forEach((button) => {
      if (button.dataset.extendedBound === '1') return;
      button.dataset.extendedBound = '1';
      button.addEventListener('click', () => document.querySelector(`[data-overview-office="${button.dataset.overviewJump}"]`)?.scrollIntoView({ behavior: 'smooth', block: 'start' }));
    });
  }
