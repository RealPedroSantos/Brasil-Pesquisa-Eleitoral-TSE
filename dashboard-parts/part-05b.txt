  function extendedPollScenarios(poll, round = 'first') {
    const scenarios = poll.rounds?.[round];
    return Array.isArray(scenarios) && scenarios.length ? scenarios : [extendedScenario('Resultado publicado', poll.candidates || [])];
  }

  function extendedPollChartMarkup(poll, round = 'first', scenarioIndex = 0) {
    const scenarios = extendedPollScenarios(poll, round);
    const safeIndex = Math.max(0, Math.min(Number(scenarioIndex) || 0, scenarios.length - 1));
    const scenario = scenarios[safeIndex];
    const people = scenario.candidates.map(extendedPerson);
    const scale = Math.max(10, ...people.map((person) => Number(person.value) || 0));
    return `<figure data-office-chart-round="${round}" data-office-chart-scenario="${safeIndex}" aria-label="${escapeHtml(poll.stateName)} — ${round === 'second' ? 'segundo turno' : 'primeiro turno'} — ${escapeHtml(scenario.label)}">

  function bindExtendedOfficeControls(root = document) {
    bindPhotoFallbacks(root);
    qsa('[data-extended-office-view]', root).forEach((button) => {
      if (button.dataset.extendedBound === '1') return;
      button.dataset.extendedBound = '1';
      button.addEventListener('click', () => showView(button.dataset.extendedOfficeView));
    });
    qsa('[data-overview-jump]', root).forEach((button) => {
      if (button.dataset.extendedBound === '1') return;
      button.dataset.extendedBound = '1';
      button.addEventListener('click', () => document.querySelector(`[data-overview-office="${button.dataset.overviewJump}"]`)?.scrollIntoView({ behavior: 'smooth', block: 'start' }));
    });
    qsa('[data-office-poll]', root).forEach((article) => {
      if (article.dataset.pollBound === '1') return;
      article.dataset.pollBound = '1';
      const poll = extendedOfficeDefinitions.governors.polls.find((item) => item.id === article.dataset.officePoll) || extendedOfficeDefinitions.senate.polls.find((item) => item.id === article.dataset.officePoll);
      if (!poll) return;
      const chart = article.querySelector('[data-office-chart]');
      const select = article.querySelector('[data-poll-scenario]');
      const buttons = qsa('[data-poll-round]', article);
      let round = 'first';
      const render = (scenarioIndex = 0) => {
        const scenarios = extendedPollScenarios(poll, round);
        select.innerHTML = scenarios.map((scenario, index) => `<option value="${index}">${escapeHtml(scenario.label)}</option>`).join('');
        const safeIndex = Math.max(0, Math.min(Number(scenarioIndex) || 0, scenarios.length - 1));
        select.value = String(safeIndex);
        chart.innerHTML = extendedPollChartMarkup(poll, round, safeIndex);
        bindPhotoFallbacks(chart);
        buttons.forEach((button) => {
          const active = button.dataset.pollRound === round;
          button.setAttribute('aria-pressed', String(active));
          button.className = `min-h-10 border px-3 text-xs font-black uppercase tracking-wide ${active ? 'border-slate-950 bg-slate-950 text-white' : 'border-slate-300 bg-white text-slate-700'}`;
        });
      };
      buttons.forEach((button) => button.addEventListener('click', () => { round = button.dataset.pollRound; render(0); }));
      select.addEventListener('change', () => render(select.value));
    });
  }
