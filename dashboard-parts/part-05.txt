  function showView(id) {
    state.view = id;
    qsa('.view-section').forEach((section) => section.classList.toggle('hidden', section.id !== id));
    qsa('[data-view]').forEach((button) => button.classList.toggle('is-active', button.dataset.view === id));
    $('sidebar').classList.remove('translate-x-0');
    $('sidebar').classList.add('-translate-x-full', 'lg:translate-x-0');
    if (id === 'overview') renderOverview();
    if (id === 'president') renderPresident();
    if (OFFICE_VIEW[id]) renderOfficeView(id, OFFICE_VIEW[id]);
    if (id === 'registry') renderRegistry();
    if (id === 'sources') renderSources();
    if (id === 'methodology') renderMethodology();
    window.scrollTo({ top: 0, behavior: 'smooth' });
    history.replaceState(null, '', `#${id}`);
  }

  function bindGlobalEvents() {
    qsa('[data-view]').forEach((button) => button.addEventListener('click', () => showView(button.dataset.view)));
    qsa('[data-nav]').forEach((link) => link.addEventListener('click', (event) => { event.preventDefault(); showView(link.dataset.nav); }));
    $('mobileMenuButton').addEventListener('click', () => {
      $('sidebar').classList.toggle('-translate-x-full');
      $('sidebar').classList.toggle('translate-x-0');
    });
    $('closeDialog').addEventListener('click', () => $('pollDialog').close());
    $('pollDialog').addEventListener('click', (event) => { if (event.target === $('pollDialog')) $('pollDialog').close(); });
    window.addEventListener('resize', () => {
      if (state.view === 'overview') renderOverview();
      if (state.view === 'president') renderPresident();
    });
  }

  async function loadRegistry() {
    state.registryLoading = true;
    state.registryError = '';
    try {
      const response = await fetch(`/api/tse-registry-v2?ts=${Date.now()}`, { cache: 'no-store' });
      const data = await response.json();
      if (!response.ok || !data.ok) throw new Error(data.detail || data.error || 'Falha ao carregar base oficial.');
      state.registry = data.records || [];
      state.registryMeta = data.meta || null;
    } catch (error) {
      state.registryError = error.message;
      state.registry = [];
      state.registryMeta = null;
    } finally {
      state.registryLoading = false;
      if (state.view === 'overview') renderOverview();
      if (OFFICE_VIEW[state.view]) renderOfficeView(state.view, OFFICE_VIEW[state.view]);
      if (state.view === 'registry') renderRegistry();
    }
  }

  function startClock() {
    const tick = () => {
      $('liveClock').textContent = new Date().toLocaleTimeString('pt-BR', { timeZone: 'America/Sao_Paulo' });
    };
    tick();
    setInterval(tick, 1000);
  }

  function init() {
    $('catalogVersion').textContent = DATA.version;
    bindGlobalEvents();
    startClock();
    const hash = location.hash.replace('#', '');
    showView(document.getElementById(hash) ? hash : 'overview');
    loadRegistry();
    requestAnimationFrame(() => $('bootScreen').remove());
  }

  document.addEventListener('DOMContentLoaded', init, { once: true });
