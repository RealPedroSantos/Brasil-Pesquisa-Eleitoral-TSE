  function trustScenarioSignature(observation) {
    return `${observation.round}|${Object.keys(observation.values || {}).sort().join('|')}`;
  }

  officeParityCurrentObservations = function officeParityComparableObservations(viewId) {
    const ui = officeParityState(viewId);
    const filtered = officeParityObservations(viewId)
      .filter((observation) => observation.uf === ui.uf)
      .filter((observation) => observation.round === ui.round)
      .filter((observation) => ui.institute === 'all' || observation.institute === ui.institute)
      .sort((a,b) => dateValue(a.date) - dateValue(b.date) || a.id.localeCompare(b.id));
    if (ui.scenario !== 'all') return filtered.filter((observation) => observation.scenario === ui.scenario);
    if (ui.round === 'first') {
      const perPoll = new Map();
      filtered.forEach((observation) => {
        const current = perPoll.get(observation.pollId);
        if (!current || observation.scenarioIndex < current.scenarioIndex) perPoll.set(observation.pollId, observation);
      });
      return [...perPoll.values()].sort((a,b) => dateValue(a.date) - dateValue(b.date) || a.id.localeCompare(b.id));
    }
    const counts = new Map();
    filtered.forEach((observation) => {
      const signature = trustScenarioSignature(observation);
      const polls = counts.get(signature) || new Set();
      polls.add(observation.pollId);
      counts.set(signature, polls);
    });
    const preferred = [...counts.entries()].sort((a,b) => b[1].size - a[1].size || a[0].localeCompare(b[0]))[0]?.[0];
    return preferred ? filtered.filter((observation) => trustScenarioSignature(observation) === preferred) : filtered;
  };

  chartMotionArrangePortraits = function chartMotionArrangePortraitsTrusted(svg) {
    const borders = qsa('.chart-candidate-photo-border', svg);
    if (!borders.length) return;
    qsa('.chart-photo-connector, .chart-photo-rail', svg).forEach((node) => node.remove());
    const fallbacks = qsa('.chart-candidate-photo-fallback', svg);
    const initials = qsa('.chart-candidate-initials', svg);
    const images = qsa('.chart-candidate-photo', svg);
    const [viewX, viewY, width, height] = chartMotionViewBox(svg);
    const largest = Math.max(...borders.map((node) => Number(node.getAttribute('width')) || 34));
    const minY = Math.max(viewY + largest/2 + 12, 54);
    const maxY = Math.max(minY, height - 112 - largest/2);
    const gap = largest + 10;
    const maxRows = Math.max(1, Math.floor((maxY - minY) / gap) + 1);
    const columns = Math.max(1, Math.min(3, Math.ceil(borders.length / maxRows)));
    const rowsPerColumn = Math.ceil(borders.length / columns);
    const columnGap = largest + 16;
    const entries = borders.map((border,index) => {
      const itemWidth = Number(border.getAttribute('width')) || 34;
      const itemHeight = Number(border.getAttribute('height')) || itemWidth;
      return { border, fallback:fallbacks[index], initials:initials[index], image:images[index], width:itemWidth, height:itemHeight, originalX:Number(border.getAttribute('x')) + itemWidth/2, originalY:Number(border.getAttribute('y')) + itemHeight/2, stroke:border.getAttribute('stroke') || '#64748b' };
    }).sort((a,b) => a.originalY - b.originalY);
    const maxPointX = Math.max(...entries.map((entry) => entry.originalX));
    const rightX = Math.min(width - largest/2 - 14, maxPointX + Math.max(68, largest + 28));
    svg.setAttribute('viewBox', `${viewX} ${viewY} ${width} ${height}`);
    svg.dataset.portraitRailInside = 'true';
    svg.dataset.portraitRailColumns = String(columns);
    for (let column=0; column<columns; column+=1) {
      const slice = entries.slice(column*rowsPerColumn, (column+1)*rowsPerColumn);
      const railX = rightX - column*columnGap;
      const rail = svgEl('line', { x1:railX, y1:minY-largest/2, x2:railX, y2:maxY+largest/2, class:'chart-photo-rail' });
      svg.insertBefore(rail, svg.querySelector('defs')?.nextSibling || svg.firstChild?.nextSibling || null);
      chartMotionSpreadY(slice, minY, maxY, gap).forEach((entry,rowIndex) => {
        const halfW = entry.width/2;
        const halfH = entry.height/2;
        const delay = 620 + column*90 + rowIndex*55;
        const targetX = railX-halfW;
        const targetY = entry.labelY-halfH;
        const connectorEndX = entry.originalX <= railX ? railX-halfW-3 : railX+halfW+3;
        const connector = svgEl('path', { d:`M ${entry.originalX} ${entry.originalY} C ${(entry.originalX+connectorEndX)/2} ${entry.originalY}, ${(entry.originalX+connectorEndX)/2} ${entry.labelY}, ${connectorEndX} ${entry.labelY}`, stroke:entry.stroke, 'stroke-width':1.6, class:'chart-photo-connector', style:`--chart-delay:${delay}ms` });
        svg.insertBefore(connector, entry.fallback || entry.border);
        [entry.fallback, entry.image, entry.border].filter(Boolean).forEach((node) => { node.setAttribute('x',targetX); node.setAttribute('y',targetY); node.setAttribute('data-chart-motion','portrait'); node.style.setProperty('--chart-delay',`${delay}ms`); });
        if (entry.image) entry.image.removeAttribute('clip-path');

        entry.border.dataset.originalPointX = String(entry.originalX);
        entry.border.dataset.originalPointY = String(entry.originalY);
        entry.border.dataset.railColumn = String(column);
      });
