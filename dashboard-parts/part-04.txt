  function registryTableMarkup(records) {
    if (state.registryLoading) return '<div class="p-8 text-center text-sm text-slate-500">Carregando base oficial do TSE…</div>';
    if (state.registryError) return `<div class="border-l-4 border-red-500 bg-red-500/10 p-4 text-sm text-red-100">${escapeHtml(state.registryError)}</div>`;
    if (!records.length) return '<div class="p-8 text-center text-sm text-slate-500">Nenhum registro encontrado para este filtro.</div>';
    return `<div class="overflow-x-auto"><table class="data-table"><thead><tr><th>Registro</th><th>Cargo</th><th>UF/local</th><th>Instituto</th><th>Campo</th><th>Amostra</th><th>Situação</th></tr></thead><tbody>${records.map((row) => `<tr><td><strong>${escapeHtml(row.registry || '—')}</strong></td><td>${escapeHtml(row.office || '—')}</td><td>${escapeHtml([row.uf, row.location].filter(Boolean).join(' · ') || 'Brasil')}</td><td>${escapeHtml(row.institute || '—')}</td><td>${escapeHtml(row.fieldStart || '—')}${row.fieldEnd ? ` a ${escapeHtml(row.fieldEnd)}` : ''}</td><td>${escapeHtml(row.sample || '—')}</td><td><span class="table-badge">${escapeHtml(row.status || 'Registrada')}</span></td></tr>`).join('')}</tbody></table></div>`;
  }

  function filteredRegistry() {
    const query = state.registrySearch.trim().toLowerCase();
    return state.registry.filter((row) => {
      const officeMatch = state.registryOffice === 'Todos' || row.office === state.registryOffice;
      const ufMatch = state.registryUf === 'Todas' || row.uf === state.registryUf;
      const textMatch = !query || [row.registry, row.office, row.uf, row.location, row.institute, row.company].join(' ').toLowerCase().includes(query);
      return officeMatch && ufMatch && textMatch;
    });
  }

    const records = filteredRegistry();
    const offices = unique(state.registry.map((row) => row.office)).sort();
    const ufs = unique(state.registry.map((row) => row.uf)).sort();

      <article class="news-card">
        <div class="grid gap-3 border-b border-slate-800 bg-slate-950 p-4 md:grid-cols-[minmax(220px,1fr)_220px_180px_auto]">
          <label class="control-label"><span>Buscar</span><input id="registrySearch" type="search" value="${escapeHtml(state.registrySearch)}" class="control-select" placeholder="Registro, instituto, cidade ou cargo"></label>
          <label class="control-label"><span>Cargo</span><select id="registryOffice" class="control-select"><option>Todos</option>${offices.map((office) => `<option ${state.registryOffice === office ? 'selected' : ''}>${escapeHtml(office)}</option>`).join('')}</select></label>
          <label class="control-label"><span>UF</span><select id="registryUf" class="control-select"><option>Todas</option>${ufs.map((uf) => `<option ${state.registryUf === uf ? 'selected' : ''}>${escapeHtml(uf)}</option>`).join('')}</select></label>
          <div class="flex items-end"><button id="clearRegistryFilters" type="button" class="control-button w-full">Limpar filtros</button></div>
        </div>
        <div class="news-card-heading"><div><span>Resultado do filtro</span><strong>${formatNumber(records.length)} registros</strong></div><small>Mostrando até ${state.registryLimit}</small></div>
        ${registryTableMarkup(records.slice(0, state.registryLimit))}
        ${records.length > state.registryLimit ? '<button id="loadMoreRegistry" type="button" class="m-4 min-h-11 border border-slate-700 bg-slate-900 px-5 py-3 text-sm font-black uppercase tracking-wide text-white hover:bg-slate-800">Carregar mais</button>' : ''}
      </article>`;

    $('registrySearch')?.addEventListener('input', (event) => { state.registrySearch = event.target.value; state.registryLimit = 80; renderRegistry(); });
    $('registryOffice')?.addEventListener('change', (event) => { state.registryOffice = event.target.value; state.registryLimit = 80; renderRegistry(); });
    $('registryUf')?.addEventListener('change', (event) => { state.registryUf = event.target.value; state.registryLimit = 80; renderRegistry(); });
    $('clearRegistryFilters')?.addEventListener('click', () => { state.registrySearch = ''; state.registryOffice = 'Todos'; state.registryUf = 'Todas'; state.registryLimit = 80; renderRegistry(); });
    $('loadMoreRegistry')?.addEventListener('click', () => { state.registryLimit += 100; renderRegistry(); });
  }
