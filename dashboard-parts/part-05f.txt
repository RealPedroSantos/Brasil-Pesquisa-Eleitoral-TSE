  const seatInteractiveOriginalHemicycleSvg = extendedHemicycleSvg;
  extendedHemicycleSvg = function interactiveHemicycleSvg(model) {
    const partyIndexes = new Map();
    return seatInteractiveOriginalHemicycleSvg(model).replace(/<circle([^>]*)><title>([^<]+)<\/title><\/circle>/g, (match, attributes, party) => {
      const index = (partyIndexes.get(party) || 0) + 1;
      partyIndexes.set(party, index);
      return `<circle${attributes} data-seat-party="${party}" data-seat-index="${index}" tabindex="0" role="button" aria-label="Cadeira ${index} — ${party}"><title>${party}</title></circle>`;
    });
  };

  function seatInteractivePartyControls(model) {
    const maxSeats = Math.max(1, ...model.parties.map((item) => item.seats));
    return model.parties.map((item) => {
      const share = model.total ? (item.seats / model.total) * 100 : 0;
      const color = extendedColor(item.party);
      return `<button type="button" class="seat-party-control" data-seat-party="${escapeHtml(item.party)}" data-seat-party-control data-seat-seats="${item.seats}" data-seat-share="${share.toFixed(2)}" data-seat-color="${color}" aria-pressed="false" style="--seat-party-color:${color}">
        <span class="grid grid-cols-[minmax(58px,auto)_1fr_auto] items-center gap-2">
          <strong class="truncate text-xs text-slate-800">${escapeHtml(item.party)}</strong>
          <span class="h-2 overflow-hidden bg-slate-100"><i data-seat-party-bar class="block h-full origin-left" style="width:${(item.seats / maxSeats) * 100}%;background:${color}"></i></span>
          <b class="font-display text-base text-slate-950">${item.seats}</b>

      </button>`;
    }).join('');
  }

  function seatInteractiveAnimateNumber(element, target, suffix = '') {
    if (!element || !Number.isFinite(Number(target))) return;
    if (element._seatAnimationFrame) cancelAnimationFrame(element._seatAnimationFrame);
    const value = Number(target);
    const duration = 420;
    const startedAt = performance.now();
    const reduced = window.matchMedia?.('(prefers-reduced-motion: reduce)').matches;
    const render = (current) => {
      const digits = suffix === '%' && !Number.isInteger(value) ? 1 : 0;
      element.textContent = `${Number(current).toLocaleString('pt-BR', { minimumFractionDigits: digits, maximumFractionDigits: digits })}${suffix}`;
    };
    if (reduced) {
      render(value);
      return;
    }
    render(0);
    const frame = (now) => {
      const progress = Math.min(1, (now - startedAt) / duration);
      const eased = 1 - Math.pow(1 - progress, 3);
      render(value * eased);
      if (progress < 1) element._seatAnimationFrame = requestAnimationFrame(frame);
    };
    element._seatAnimationFrame = requestAnimationFrame(frame);
  }

  function seatInteractiveReset(simulation) {
    simulation.classList.remove('has-seat-selection');
    delete simulation.dataset.seatSelected;
    qsa('circle[data-seat-party]', simulation).forEach((circle) => circle.classList.remove('is-seat-selected', 'is-seat-dimmed'));
    qsa('[data-seat-party-control]', simulation).forEach((button) => {
      button.classList.remove('is-active');
      button.setAttribute('aria-pressed', 'false');
    });
    const inspector = simulation.querySelector('[data-seat-inspector]');
    const total = Number(simulation.dataset.seatTotal || 0);
    const parties = Number(simulation.dataset.seatPartyCount || 0);
    if (inspector) {
      inspector.classList.remove('is-seat-inspector-active');
      void inspector.offsetWidth;
      inspector.classList.add('is-seat-inspector-active');
      inspector.style.borderLeftColor = '#10b981';
      inspector.querySelector('[data-seat-inspector-swatch]').style.background = '#10b981';
      inspector.querySelector('[data-seat-inspector-eyebrow]').textContent = 'Visão geral';
      inspector.querySelector('[data-seat-inspector-title]').textContent = 'Explore a composição';
      inspector.querySelector('[data-seat-inspector-rank]').textContent = String(parties);
      inspector.querySelector('[data-seat-inspector-copy]').textContent = 'Selecione uma bancada para destacar suas cadeiras e consultar posição, participação e diferença para a maior bancada.';
      seatInteractiveAnimateNumber(inspector.querySelector('[data-seat-inspector-seats]'), total);
      seatInteractiveAnimateNumber(inspector.querySelector('[data-seat-inspector-share]'), 100, '%');
    }
  }

  function seatInteractiveSelect(simulation, party) {
    if (!simulation || !party) return;
    if (simulation.dataset.seatSelected === party) {
      seatInteractiveReset(simulation);
      return;
    }

    const control = qsa('[data-seat-party-control]', simulation).find((button) => button.dataset.seatParty === party);
    if (!control) return;
    const seats = Number(control.dataset.seatSeats || 0);
    const share = Number(control.dataset.seatShare || 0);
    const color = control.dataset.seatColor || extendedColor(party);
    const controls = qsa('[data-seat-party-control]', simulation);
    const ranking = controls.slice().sort((a, b) => Number(b.dataset.seatSeats) - Number(a.dataset.seatSeats));
    const rank = ranking.findIndex((button) => button.dataset.seatParty === party) + 1;
    const leaderSeats = Number(ranking[0]?.dataset.seatSeats || 0);
    const gap = Math.max(0, leaderSeats - seats);

    simulation.dataset.seatSelected = party;
    simulation.classList.add('has-seat-selection');
    let selectedIndex = 0;
    qsa('circle[data-seat-party]', simulation).forEach((circle) => {
      const active = circle.dataset.seatParty === party;
      circle.classList.toggle('is-seat-selected', active);
      circle.classList.toggle('is-seat-dimmed', !active);
      if (active) {
        circle.style.setProperty('--seat-click-delay', `${Math.min(260, selectedIndex * 4)}ms`);
        selectedIndex += 1;
      }
    });
    controls.forEach((button) => {
      const active = button.dataset.seatParty === party;
      button.classList.toggle('is-active', active);
      button.setAttribute('aria-pressed', String(active));
    });

    const inspector = simulation.querySelector('[data-seat-inspector]');
    if (!inspector) return;
    inspector.classList.remove('is-seat-inspector-active');
    void inspector.offsetWidth;
    inspector.classList.add('is-seat-inspector-active');
    inspector.style.borderLeftColor = color;
    inspector.querySelector('[data-seat-inspector-swatch]').style.background = color;
    inspector.querySelector('[data-seat-inspector-eyebrow]').textContent = `Bancada selecionada · ${rank}ª posição`;
    inspector.querySelector('[data-seat-inspector-title]').textContent = party;
    inspector.querySelector('[data-seat-inspector-rank]').textContent = `#${rank}`;
    inspector.querySelector('[data-seat-inspector-copy]').textContent = party === 'Sem projeção'
      ? 'Estas cadeiras representam apenas o tamanho total da casa. Ainda não existe distribuição partidária verificável para este modelo.'
      : gap === 0
        ? `${party} aparece como a maior bancada deste modelo metodológico.`
        : `${party} está a ${gap} cadeira${gap === 1 ? '' : 's'} da maior bancada simulada.`;
    seatInteractiveAnimateNumber(inspector.querySelector('[data-seat-inspector-seats]'), seats);
    seatInteractiveAnimateNumber(inspector.querySelector('[data-seat-inspector-share]'), share, '%');
  }

  function seatInteractiveBind(simulation) {
    if (!simulation || simulation.dataset.seatInteractiveBound === 'true') return;
    simulation.dataset.seatInteractiveBound = 'true';
    simulation.addEventListener('click', (event) => {
      const reset = event.target.closest?.('[data-seat-reset]');
      if (reset) {
        seatInteractiveReset(simulation);
        return;
      }
      const target = event.target.closest?.('[data-seat-party]');
      if (target) seatInteractiveSelect(simulation, target.dataset.seatParty);
    });
    simulation.addEventListener('keydown', (event) => {
      if (event.key !== 'Enter' && event.key !== ' ') return;
      const target = event.target.closest?.('circle[data-seat-party]');
      if (!target) return;
      event.preventDefault();
      seatInteractiveSelect(simulation, target.dataset.seatParty);
    });
  }

    seatInteractivePreviousEnhanceSimulation(simulation);

  };
