  function bindExplorerControls(host, options) {
    host.querySelectorAll('[data-round]').forEach((button) => button.addEventListener('click', () => {
      state.round = Number(button.dataset.round);
      state.institute = 'all';
      if (state.round === 2) {
        const available = DATA.polls.find((poll) => poll.round === 2)?.matchup;
        if (!DATA.polls.some((poll) => poll.round === 2 && poll.matchup === state.matchup)) state.matchup = available;
      }
      refreshPollViews();
    }));

    host.querySelector('#instituteFilter')?.addEventListener('change', (event) => {
      state.institute = event.target.value;
      refreshPollViews();
    });

    host.querySelector('#matchupFilter')?.addEventListener('change', (event) => {
      state.matchup = event.target.value;
      state.institute = 'all';
      refreshPollViews();
    });

    host.querySelectorAll('[data-candidate]').forEach((button) => button.addEventListener('click', () => {
      const id = button.dataset.candidate;
      if (state.selectedCandidates.has(id)) state.selectedCandidates.delete(id);
      else state.selectedCandidates.add(id);
      if (!state.selectedCandidates.size) state.selectedCandidates.add(id);
      refreshPollViews();
    }));

    host.querySelector('#showAllCandidates')?.addEventListener('click', () => {
      currentPolls().flatMap((poll) => Object.keys(poll.values)).forEach((id) => state.selectedCandidates.add(id));
      refreshPollViews();
    });

    host.querySelector('#openLatestPoll')?.addEventListener('click', () => {
      const latest = currentPolls().at(-1);
      if (latest) openPollDialog(latest.id);
    });
  }

  function refreshPollViews() {
    if (state.view === 'overview') renderOverview();
    if (state.view === 'president') renderPresident();
  }

  function drawPollChart(svg, tooltip, legend) {
    if (!svg) return;
    const polls = currentPolls();
    const candidateIds = visibleCandidateIds(polls);
    const width = Math.max(900, polls.length * 76 + 140);
    const height = 470;
    const pad = { left: 54, right: 38, top: 30, bottom: 92 };
    const plotW = width - pad.left - pad.right;
    const plotH = height - pad.top - pad.bottom;
    const maxY = 60;
    const x = (index) => pad.left + (polls.length <= 1 ? plotW / 2 : index / (polls.length - 1) * plotW);
    const y = (value) => pad.top + plotH - (Number(value) / maxY) * plotH;

    svg.replaceChildren();
    svg.setAttribute('viewBox', `0 0 ${width} ${height}`);

    for (let value = 0; value <= maxY; value += 10) {
      const yy = y(value);
      svg.appendChild(svgEl('line', { x1: pad.left, y1: yy, x2: width - pad.right, y2: yy, class: 'chart-grid-line' }));
      svg.appendChild(svgEl('text', { x: pad.left - 9, y: yy + 4, class: 'chart-axis-label', 'text-anchor': 'end' }, `${value}%`));
    }

    polls.forEach((poll, index) => {
      const xx = x(index);
      svg.appendChild(svgEl('line', { x1: xx, y1: pad.top, x2: xx, y2: pad.top + plotH, class: 'chart-event-line' }));
      const dateText = svgEl('text', { x: xx, y: height - 55, class: 'chart-date-label', 'text-anchor': 'middle' }, formatDate(poll.date).toUpperCase());
