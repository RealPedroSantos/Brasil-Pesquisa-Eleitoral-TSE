  function dataMotionParseNumber(text) {
    const raw = String(text || '').trim();
    const match = raw.match(/^(\d{1,3}(?:\.\d{3})*|\d+)(?:,(\d+))?\s*(%|cadeiras?|vagas?|registros?|cenários?|pesquisas?|perfis?)?$/i);
    if (!match) return null;
    const integer = match[1].replace(/\./g, '');
    const decimals = match[2] || '';
    const value = Number(`${integer}${decimals ? `.${decimals}` : ''}`);
    if (!Number.isFinite(value)) return null;
    return {
      value,
      digits: decimals.length,
      suffix: match[3] ? ` ${match[3]}` : raw.endsWith('%') ? '%' : ''
    };
  }

  function dataMotionPrepareCount(element) {
    if (!element || element.dataset.countPrepared === 'true') return;
    const parsed = dataMotionParseNumber(element.textContent);
    if (!parsed) return;
    element.dataset.countPrepared = 'true';
    element.dataset.countTarget = String(parsed.value);
    element.dataset.countDigits = String(parsed.digits);
    element.dataset.countSuffix = parsed.suffix;
    element.dataset.countOriginal = element.textContent.trim();
  }

  function dataMotionCountUp(element) {
    if (!element || element.dataset.countAnimated === 'true') return;
    const target = Number(element.dataset.countTarget);
    if (!Number.isFinite(target)) return;
    element.dataset.countAnimated = 'true';
    const digits = Number(element.dataset.countDigits || 0);
    const suffix = element.dataset.countSuffix || '';
    const duration = Math.min(1100, Math.max(420, target * 4));
    const startedAt = performance.now();
    const reduced = window.matchMedia?.('(prefers-reduced-motion: reduce)').matches;

    const renderValue = (value) => {
      const formatted = Number(value).toLocaleString('pt-BR', {
        minimumFractionDigits: digits,
        maximumFractionDigits: digits
      });
      element.textContent = `${formatted}${suffix}`;
    };

    if (reduced) {
      renderValue(target);
      element.classList.add('is-count-complete');
      return;
    }

    renderValue(0);
    const frame = (now) => {
      const progress = Math.min(1, (now - startedAt) / duration);
      const eased = 1 - Math.pow(1 - progress, 3);
      renderValue(target * eased);
      if (progress < 1) requestAnimationFrame(frame);
      else {
        element.textContent = element.dataset.countOriginal;
        element.classList.add('is-count-complete');
      }
    };
    requestAnimationFrame(frame);
  }

  function dataMotionReveal(element) {
    if (!element || element.classList.contains('is-data-visible')) return;
    element.classList.add('is-data-visible');
    if (element.matches('[data-count-target]')) dataMotionCountUp(element);
    qsa('[data-count-target]', element).forEach(dataMotionCountUp);
  }

  const dataMotionObserver = 'IntersectionObserver' in window
    ? new IntersectionObserver((entries, observer) => {
        entries.forEach((entry) => {
          if (!entry.isIntersecting) return;
          dataMotionReveal(entry.target);
          observer.unobserve(entry.target);
        });
      }, { threshold: 0.08, rootMargin: '0px 0px -4% 0px' })
    : null;

  function dataMotionMark(element, delay = 0) {
    if (!element || element.dataset.dataMotion) return;
    element.dataset.dataMotion = 'reveal';
    element.style.setProperty('--data-delay', `${Math.min(650, delay)}ms`);
    if (dataMotionObserver) dataMotionObserver.observe(element);
    else dataMotionReveal(element);
  }

  function dataMotionEnhanceSimulation(simulation) {
    if (!simulation || simulation.dataset.simulationAnimated === 'true') return;
    simulation.dataset.simulationAnimated = 'true';

    [...simulation.children].forEach((child, index) => dataMotionMark(child, index * 90));
    qsa('svg, details, [class*="grid-cols"], [class*="inline-flex"]', simulation)
      .forEach((element, index) => dataMotionMark(element, 130 + index * 28));

    qsa('b, strong', simulation).forEach(dataMotionPrepareCount);
    qsa('[data-count-target]', simulation).forEach((element, index) => {
      dataMotionMark(element, 180 + index * 34);
      if (!dataMotionObserver) dataMotionCountUp(element);
    });
  }

  function dataMotionEnhance(root = document) {
    const scope = root.nodeType === 1 ? root : document;
    const simulations = [];
    if (scope.matches?.('[data-seat-simulation]')) simulations.push(scope);
    simulations.push(...qsa('[data-seat-simulation]', scope));
    simulations.forEach(dataMotionEnhanceSimulation);

    const selectors = [
      '.chart-ranking-row',
      '.candidate-roster-card',
      '.poll-result-chip',
      '.data-table tbody tr',
      '[data-office-chart-candidate]',
      '.status-chip',
      '.poll-dialog-candidate',
      '.registry-document-field'
    ];

    selectors.forEach((selector) => {
      const elements = [];
      if (scope.matches?.(selector)) elements.push(scope);
      elements.push(...qsa(selector, scope));
      elements.forEach((element, index) => dataMotionMark(element, (index % 14) * 34));
    });

    qsa('.chart-ranking-row > b, .poll-result-chip b, .poll-dialog-candidate > b, [data-office-chart-candidate] > b, [class*="text-4xl"][class*="font-display"]', scope)
      .forEach((element, index) => {
        dataMotionPrepareCount(element);
        if (element.dataset.countTarget) dataMotionMark(element, 100 + (index % 12) * 30);
      });
  }

  function registryDocumentKey(row) {
    const source = [row.registry, row.office, row.uf, row.location, row.institute, row.company, row.fieldStart, row.fieldEnd, row.publication, row.sample]
      .map((value) => String(value || '').trim().toUpperCase()).join('|');
    let hash = 2166136261;
    for (let index = 0; index < source.length; index += 1) {
      hash ^= source.charCodeAt(index);
      hash = Math.imul(hash, 16777619);
    }
    return `tse-${(hash >>> 0).toString(36)}`;
  }

  function registryDocumentField(label, value, index) {
    if (value === undefined || value === null || String(value).trim() === '') return '';
    return `<div class="registry-document-field border border-slate-200 bg-slate-50 p-3" style="--document-delay:${Math.min(520, index * 38)}ms">
      <span class="text-[10px] font-bold uppercase tracking-[0.14em] text-slate-500">${escapeHtml(label)}</span>
      <strong class="mt-1 block break-words text-sm text-slate-950">${escapeHtml(value)}</strong>
    </div>`;
  }

  function registryDocumentBindEvents() {
    document.addEventListener('click', (event) => {
      const target = event.target.closest?.('[data-registry-document]');
      if (!target) return;
      const key = target.dataset.registryDocument;
      if (!registryDocumentStore.has(key)) return;
      event.preventDefault();
      openRegistryDocument(key);
    });

    document.addEventListener('keydown', (event) => {
      if (event.key !== 'Enter' && event.key !== ' ') return;
      const target = event.target.closest?.('tr[data-registry-document]');
      if (!target || event.target.closest('button')) return;
      event.preventDefault();
      openRegistryDocument(target.dataset.registryDocument);
    });
  }

  function dataMotionInit() {
    dataMotionInstallStyles();
    registryDocumentBindEvents();
    dataMotionEnhance(document);

    const mutationObserver = new MutationObserver((mutations) => {
      mutations.forEach((mutation) => {
        mutation.addedNodes.forEach((node) => {
          if (node.nodeType === 1) dataMotionEnhance(node);
        });
      });
    });
    mutationObserver.observe(document.body, { childList: true, subtree: true });
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', dataMotionInit, { once: true });
  else dataMotionInit();
