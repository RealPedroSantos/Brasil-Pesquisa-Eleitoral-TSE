
  // Correção visual final: preserva a largura real do SVG e separa observações da mesma data.
  function verifiedChartClusterLayout(items) {
    const groups = new Map();
    items.forEach((item) => {
      const timestamp = verifiedTimelineDateMs(item);
      if (!Number.isFinite(timestamp)) return;
      if (!groups.has(timestamp)) groups.set(timestamp, []);
      groups.get(timestamp).push(item);
    });

    const largestCluster = Math.max(1, ...[...groups.values()].map((group) => group.length));
    const clusterStep = largestCluster > 8 ? 18 : largestCluster > 5 ? 22 : 28;
    const offsetByItem = new Map();

    groups.forEach((group) => {
      group
        .slice()
        .sort((a, b) => String(a.id || '').localeCompare(String(b.id || ''), 'pt-BR'))
        .forEach((item, index, sorted) => {
          offsetByItem.set(item, (index - (sorted.length - 1) / 2) * clusterStep);
        });
    });

    return {
      groups,
      offsetByItem,
      clusterStep,
      largestCluster,
      maxClusterHalf: ((largestCluster - 1) * clusterStep) / 2
    };
  }

  verifiedTimelineRender = function verifiedTimelineRenderLayoutFixed({ svg, tooltip, legend, ranking, items, candidateIds, candidateFor, round, openItem, tooltipFor, rankingOpenAttribute }) {
    if (!svg) return;

    const validItems = items
      .filter((item) => Number.isFinite(verifiedTimelineDateMs(item)))
      .slice()
      .sort((a, b) => verifiedTimelineDateMs(a) - verifiedTimelineDateMs(b) || String(a.id || '').localeCompare(String(b.id || ''), 'pt-BR'));
    const dates = unique(validItems.map(verifiedTimelineDateMs)).sort((a, b) => a - b);
    const dateMin = dates[0] || Date.now();
    const dateMax = dates.at(-1) || dateMin;
    const clusterLayout = verifiedChartClusterLayout(validItems);
    const repeatedObservations = Math.max(0, validItems.length - dates.length);
    const width = Math.max(960, Math.max(1, dates.length) * 170 + repeatedObservations * clusterLayout.clusterStep + 140);
    const height = candidateIds.length > 10 ? 560 : 500;
    const pad = { left: 62, right: 42, top: 42, bottom: 92 };
    const plotW = width - pad.left - pad.right;
    const plotH = height - pad.top - pad.bottom;
    const rawMax = Math.max(0, ...validItems.flatMap((item) => Object.values(item.values || {}).map(Number).filter(Number.isFinite)));
    const maxY = Math.max(20, Math.ceil((rawMax + 5) / 10) * 10);
    const axisLeft = pad.left + clusterLayout.maxClusterHalf;
    const axisRight = width - pad.right - clusterLayout.maxClusterHalf;
    const axisWidth = Math.max(1, axisRight - axisLeft);
    const dateX = (timestamp) => dateMax === dateMin
      ? pad.left + plotW / 2
      : axisLeft + ((timestamp - dateMin) / (dateMax - dateMin)) * axisWidth;
    const x = (item) => dateX(verifiedTimelineDateMs(item)) + (clusterLayout.offsetByItem.get(item) || 0);
    const y = (value) => pad.top + plotH - (Number(value) / maxY) * plotH;

    svg.replaceChildren();
    svg.setAttribute('viewBox', `0 0 ${width} ${height}`);
    svg.setAttribute('width', String(width));
    svg.setAttribute('height', String(height));
    svg.style.width = `max(100%, ${width}px)`;
    svg.style.minWidth = `${width}px`;
    svg.style.maxWidth = 'none';
    svg.style.height = 'auto';
    svg.style.aspectRatio = `${width} / ${height}`;
    svg.dataset.timelineValidation = 'verified-v2';
    svg.dataset.timelineScale = 'proportional-date';
    svg.dataset.portraitRailInside = 'disabled';
    svg.dataset.sameDateLayout = 'clustered';
    svg.dataset.intrinsicWidth = 'preserved';

    const defs = svgEl('defs');
    const clipId = `${svg.id || 'chart'}-verified-plot`;
    const clip = svgEl('clipPath', { id: clipId });
    clip.appendChild(svgEl('rect', { x: pad.left, y: pad.top, width: plotW, height: plotH }));
    defs.appendChild(clip);
    svg.appendChild(defs);

    for (let value = 0; value <= maxY; value += 10) {
      const yy = y(value);
      svg.appendChild(svgEl('line', { x1: pad.left, y1: yy, x2: width - pad.right, y2: yy, class: 'chart-grid-line' }));
      svg.appendChild(svgEl('text', { x: pad.left - 9, y: yy + 4, class: 'chart-axis-label', 'text-anchor': 'end' }, `${value}%`));
    }

    dates.forEach((timestamp) => {
      const group = clusterLayout.groups.get(timestamp) || [];
      const item = group[0];
      if (!item) return;
      const xx = dateX(timestamp);
      svg.appendChild(svgEl('line', { x1: xx, y1: pad.top, x2: xx, y2: pad.top + plotH, class: 'chart-event-line' }));
      const label = item.publication || new Date(timestamp).toLocaleDateString('pt-BR');
      svg.appendChild(svgEl('text', { x: xx, y: height - 38, class: 'chart-date-label', 'text-anchor': 'middle' }, label));
      if (group.length > 1) {
        svg.appendChild(svgEl('text', { x: xx, y: height - 21, class: 'chart-scenario-label', 'text-anchor': 'middle' }, `${group.length} cenários`));
      }
    });

    const plot = svgEl('g', { 'clip-path': `url(#${clipId})` });
    svg.appendChild(plot);

    candidateIds.forEach((candidateId, candidateIndex) => {
      const candidate = candidateFor(candidateId);
      if (!candidate) return;
      const candidateItems = validItems.filter((item) => Object.hasOwn(item.values || {}, candidateId));
      const seriesGroups = new Map();

      candidateItems.forEach((item) => {
        const key = verifiedTimelineScenarioKey(item);
        if (!seriesGroups.has(key)) seriesGroups.set(key, []);
        seriesGroups.get(key).push(item);
      });

      seriesGroups.forEach((groupItems) => {
        const sorted = groupItems.slice().sort((a, b) => verifiedTimelineDateMs(a) - verifiedTimelineDateMs(b) || String(a.id || '').localeCompare(String(b.id || ''), 'pt-BR'));
        const uniqueDates = new Set(sorted.map(verifiedTimelineDateMs));
        if (sorted.length > 1 && uniqueDates.size === sorted.length) {
          plot.appendChild(svgEl('path', {
            d: sorted.map((item, index) => `${index ? 'L' : 'M'} ${x(item)} ${y(item.values[candidateId])}`).join(' '),
            fill: 'none',
            stroke: candidate.color,
            'stroke-width': round === 2 || round === 'second' ? 5 : 3.5,
            'stroke-linecap': 'round',
            'stroke-linejoin': 'round',
            class: 'chart-series-line',
            'data-series-scenario': verifiedTimelineScenarioKey(sorted[0]),
            'data-chart-motion': 'line',
            pathLength: '1',
            style: `--chart-delay:${candidateIndex * 70}ms`
          }));
        }
      });

      candidateItems.forEach((item, pointIndex) => {
        const value = Number(item.values[candidateId]);
        const xx = x(item);
        const yy = y(value);
        const dot = svgEl('circle', {
          cx: xx,
          cy: yy,
          r: round === 2 || round === 'second' ? 6.5 : 5.5,
          fill: candidate.color,
          stroke: '#020617',
          'stroke-width': 3,
          class: 'chart-point',
          'data-chart-motion': 'point',
          style: `--chart-delay:${Math.min(760, 100 + pointIndex * 24)}ms`
        });
        const hit = svgEl('circle', {
          cx: xx,
          cy: yy,
          r: 14,
          fill: 'transparent',
          class: 'chart-hit-area',
          tabindex: '0',
          role: 'button',
          'aria-label': `${candidate.name}, ${formatNumber(value)}%, ${item.institute}, ${item.publication}`
        });
        const show = (event) => tooltipFor(event, candidate, item, value);
        const showFromKeyboard = () => {
          const rect = svg.getBoundingClientRect();
          tooltipFor({
            clientX: rect.left + (xx / width) * rect.width,
            clientY: rect.top + (yy / height) * rect.height
          }, candidate, item, value);
        };
        hit.addEventListener('pointerenter', show);
        hit.addEventListener('pointermove', show);
        hit.addEventListener('pointerleave', () => tooltip?.classList.add('hidden'));
        hit.addEventListener('focus', showFromKeyboard);
        hit.addEventListener('blur', () => tooltip?.classList.add('hidden'));
        hit.addEventListener('click', () => openItem(item.id));
        hit.addEventListener('keydown', (event) => {
          if (event.key === 'Enter' || event.key === ' ') {
            event.preventDefault();
            openItem(item.id);
          }
        });
        plot.appendChild(dot);
        plot.appendChild(hit);
      });
    });

    if (!validItems.length) {
      svg.appendChild(svgEl('text', { x: width / 2, y: height / 2, class: 'chart-empty-label', 'text-anchor': 'middle' }, 'Nenhuma pesquisa nominal verificável para este filtro.'));
    }

    const rankingRows = candidateIds.map((id) => {
      const candidate = candidateFor(id);
      const item = validItems.slice().reverse().find((entry) => Object.hasOwn(entry.values || {}, id));
      return candidate && item ? { candidate, item, value: item.values[id] } : null;
    }).filter(Boolean).sort((a, b) => b.value - a.value || a.candidate.name.localeCompare(b.candidate.name, 'pt-BR'));

    if (ranking) {
      ranking.innerHTML = `<div class="chart-ranking-heading"><span>Último dado disponível</span><strong>Ranking com fotos</strong></div><div class="chart-ranking-list">${rankingRows.map((row, index) => `<button type="button" ${rankingOpenAttribute}="${escapeHtml(row.item.id)}" class="chart-ranking-row" style="--candidate-color:${row.candidate.color}"><span class="chart-ranking-position">${index + 1}</span>${candidatePhotoMarkup(row.candidate, 'sm')}<span class="min-w-0"><strong>${escapeHtml(row.candidate.name)}</strong><small>${escapeHtml(row.candidate.party)} · ${escapeHtml(row.item.publication)}</small></span><b>${formatNumber(row.value)}%</b></button>`).join('') || '<div class="p-5 text-sm text-slate-500">Sem ranking publicado para este filtro.</div>'}</div>`;
      ranking.querySelectorAll(`[${rankingOpenAttribute}]`).forEach((button) => button.addEventListener('click', () => openItem(button.getAttribute(rankingOpenAttribute))));
      bindPhotoFallbacks(ranking);
    }

    if (legend) {
      legend.innerHTML = candidateIds.map((id) => {
        const candidate = candidateFor(id);
        return candidate ? `<span class="candidate-legend-item" style="--candidate-color:${candidate.color}">${candidatePhotoMarkup(candidate, 'xs')}<span>${escapeHtml(candidate.name)} · ${escapeHtml(candidate.party)}</span></span>` : '';
      }).join('');
      bindPhotoFallbacks(legend);
    }
  };

  const verifiedLayoutOriginalOfficeParityDrawChart = officeParityDrawChart;
  officeParityDrawChart = function officeParityDrawChartWithLayoutFix(root, viewId) {
    verifiedLayoutOriginalOfficeParityDrawChart(root, viewId);
    const note = root.querySelector('[data-office-parity-explorer] .border-t.border-slate-800 p');
    if (note) note.textContent = 'O eixo horizontal respeita o intervalo real entre as datas. Cenários publicados no mesmo dia são separados horizontalmente, e as linhas só conectam observações realmente comparáveis.';
  };
